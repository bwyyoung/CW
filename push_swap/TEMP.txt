void		merge(t_push_swap *ps, int len_h1, int len_h2)
{
	while (ps->i < len_h1 && ps->j < len_h2)
	{
		ps_verbose("check piles |%i|%i|%i\n", ps->pile_a->num , ps->pile_b->num,
				  (ps->pile_a->num > ps->pile_b->num));
		if (ps->pile_a->num <= ps->pile_b->num)
		{
			op_ra_n(ps, 1);
			ps->i++;
			ps_verbose("option 1\n");
			continue ;
		}
		ps_verbose("option 2\n");
		merge_pile_b(ps);
		ps->j++;
	}
	ps->i--;
	ps->j--;
	while (++ps->i < len_h1)
		op_ra_n(ps, 1);
	while (++ps->j < len_h2)
		merge_pile_b(ps);
}

	*len_h1 = 0;
